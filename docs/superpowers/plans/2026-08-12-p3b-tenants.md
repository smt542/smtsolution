create or replace function save_unit_occupancy(
  p_building_id bigint, p_unit_id bigint, p_occupancy_id bigint,
  p_unit jsonb, p_occupancy jsonb, p_meters jsonb, p_accounts jsonb, p_lease jsonb
) returns jsonb language plpgsql security definer set search_path = public as $$
declare v_org bigint; v_unit_id bigint := p_unit_id; v_occ_id bigint := p_occupancy_id; v_item jsonb;
begin
  v_org := current_user_org();
  if v_org is null then raise exception 'no org for caller'; end if;
  if not current_user_has_module('tenants') then raise exception 'forbidden: tenants required'; end if;
  if not exists (select 1 from buildings where id = p_building_id and org_id = v_org) then
    raise exception 'building not in org'; end if;

  if v_unit_id is null then
    insert into units(building_id, unit_no, dong, area_pyeong, room_count)
      values (p_building_id, p_unit->>'unit_no', p_unit->>'dong',
              nullif(p_unit->>'area_pyeong','')::numeric, nullif(p_unit->>'room_count','')::int)
      returning id into v_unit_id;
  else
    update units set unit_no = coalesce(p_unit->>'unit_no', unit_no), dong = p_unit->>'dong',
      area_pyeong = nullif(p_unit->>'area_pyeong','')::numeric, room_count = nullif(p_unit->>'room_count','')::int
      where id = v_unit_id and building_id = p_building_id;
    if not found then raise exception 'unit not in building'; end if;
  end if;

  if v_occ_id is null then
    insert into occupancies(unit_id, tenant_name, payer_name, phone, phone2, phone3, vehicle_no,
      resident_count, room_type_id, move_in_date, move_out_date, status,
      billing_enabled, print_enabled, late_fee_enabled, single_print, billing_note, settlement_note, facility_note)
      values (v_unit_id, p_occupancy->>'tenant_name', p_occupancy->>'payer_name', p_occupancy->>'phone',
        p_occupancy->>'phone2', p_occupancy->>'phone3', p_occupancy->>'vehicle_no',
        nullif(p_occupancy->>'resident_count','')::int, nullif(p_occupancy->>'room_type_id','')::bigint,
        nullif(p_occupancy->>'move_in_date','')::date, nullif(p_occupancy->>'move_out_date','')::date,
        coalesce(p_occupancy->>'status','occupied')::occupancy_status,
        coalesce((p_occupancy->>'billing_enabled')::boolean,true), coalesce((p_occupancy->>'print_enabled')::boolean,true),
        coalesce((p_occupancy->>'late_fee_enabled')::boolean,true), coalesce((p_occupancy->>'single_print')::boolean,false),
        p_occupancy->>'billing_note', p_occupancy->>'settlement_note', p_occupancy->>'facility_note')
      returning id into v_occ_id;
  else
    update occupancies set tenant_name = p_occupancy->>'tenant_name', payer_name = p_occupancy->>'payer_name',
      phone = p_occupancy->>'phone', phone2 = p_occupancy->>'phone2', phone3 = p_occupancy->>'phone3',
      vehicle_no = p_occupancy->>'vehicle_no', resident_count = nullif(p_occupancy->>'resident_count','')::int,
      room_type_id = nullif(p_occupancy->>'room_type_id','')::bigint,
      move_in_date = nullif(p_occupancy->>'move_in_date','')::date, move_out_date = nullif(p_occupancy->>'move_out_date','')::date,
      status = coalesce(p_occupancy->>'status','occupied')::occupancy_status,
      billing_enabled = coalesce((p_occupancy->>'billing_enabled')::boolean,true),
      print_enabled = coalesce((p_occupancy->>'print_enabled')::boolean,true),
      late_fee_enabled = coalesce((p_occupancy->>'late_fee_enabled')::boolean,true),
      single_print = coalesce((p_occupancy->>'single_print')::boolean,false),
      billing_note = p_occupancy->>'billing_note', settlement_note = p_occupancy->>'settlement_note',
      facility_note = p_occupancy->>'facility_note'
      where id = v_occ_id and unit_id = v_unit_id;
    if not found then raise exception 'occupancy not in unit'; end if;
  end if;

  for v_item in select value from jsonb_array_elements(coalesce(p_meters,'[]'::jsonb)) loop
    insert into unit_meters(unit_id, utility, meter_no, customer_no)
      values (v_unit_id, v_item->>'utility', v_item->>'meter_no', v_item->>'customer_no')
      on conflict (unit_id, utility) do update set meter_no = excluded.meter_no, customer_no = excluded.customer_no;
  end loop;
  for v_item in select value from jsonb_array_elements(coalesce(p_accounts,'[]'::jsonb)) loop
    insert into unit_accounts(unit_id, kind, bank, account, holder, cms_member_no)
      values (v_unit_id, v_item->>'kind', v_item->>'bank', v_item->>'account', v_item->>'holder', v_item->>'cms_member_no')
      on conflict (unit_id, kind) do update set bank = excluded.bank, account = excluded.account,
        holder = excluded.holder, cms_member_no = excluded.cms_member_no;
  end loop;
  if p_lease is not null and p_lease <> 'null'::jsonb then
    insert into lease_contracts(occupancy_id, deposit, monthly_rent, start_date, end_date, expiry_alert)
      values (v_occ_id, nullif(p_lease->>'deposit','')::bigint, nullif(p_lease->>'monthly_rent','')::bigint,
        nullif(p_lease->>'start_date','')::date, nullif(p_lease->>'end_date','')::date,
        coalesce((p_lease->>'expiry_alert')::boolean,false))
      on conflict (occupancy_id) do update set deposit = excluded.deposit, monthly_rent = excluded.monthly_rent,
        start_date = excluded.start_date, end_date = excluded.end_date, expiry_alert = excluded.expiry_alert;
  end if;

  return jsonb_build_object('unit_id', v_unit_id, 'occupancy_id', v_occ_id);
end $$;
revoke execute on function save_unit_occupancy(bigint,bigint,bigint,jsonb,jsonb,jsonb,jsonb,jsonb) from anon;

-- ===== 목록 RPC =====
create or replace function list_tenants(p_building_id bigint)
returns table(
  unit_id bigint, occupancy_id bigint, unit_no text, tenant_name text,
  room_type_label text, resident_count int, move_in_date date,
  is_vacant boolean, is_overdue boolean
) language sql stable security definer set search_path = public as $$
  select u.id, o.id, u.unit_no, o.tenant_name, rt.label, o.resident_count, o.move_in_date,
    coalesce(o.status = 'vacant', true) as is_vacant,
    false as is_overdue
  from units u
  join buildings b on b.id = u.building_id and b.org_id = current_user_org()
  left join lateral (
    select * from occupancies oc where oc.unit_id = u.id and oc.deleted_at is null
    order by oc.created_at desc limit 1
  ) o on true
  left join room_types rt on rt.id = o.room_type_id
  where u.building_id = p_building_id and u.deleted_at is null
  order by u.unit_no
$$;
revoke execute on function list_tenants(bigint) from anon;

export async function getUnitOccupancyFull(unitId: number): Promise<UnitOccupancyFull> {
  const sb = createClient();
  const { data: u, error: ue } = await sb.from("units").select("*").eq("id", unitId).is("deleted_at", null).single();
  if (ue) throw new Error(ue.message);
  const { data: o, error: oe } = await sb.from("occupancies").select("*").eq("unit_id", unitId).is("deleted_at", null)
    .order("created_at", { ascending: false }).limit(1).maybeSingle();
  if (oe) throw new Error(oe.message);
  const occupancy = o ? rowToOccupancy(o) : null;
  const [meters, accounts, history, lease] = await Promise.all([
    listMetersByUnit(unitId), listAccountsByUnit(unitId), listHistoryByUnit(unitId),
    occupancy ? getLeaseByOccupancy(occupancy.id) : Promise.resolve(null),
  ]);
  return { unit: rowToUnit(u), occupancy, meters, accounts, lease, history };
}

"use client";
import React, { useEffect, useMemo, useState } from "react";
import { useRouter } from "next/navigation";
import { toast } from "sonner";
import { Button } from "@/components/common";
import { useSaveUnitOccupancy, useUnitOccupancyFull, type SaveUnitOccupancyPayload } from "@/hooks/reactquery/occupancies";
import { useRoomTypes } from "@/hooks/reactquery/roomTypes";
import { occupancyBasicSchema } from "@/components/forms/schema/occupancy";
import { BasicInfoSection, type TenantBasicValue } from "./tenant-sections/BasicInfoSection";
import { BillingSettingsSection, type TenantBillingValue } from "./tenant-sections/BillingSettingsSection";
import { MetersAccountsSection, type MeterRow, type AccountRow } from "./tenant-sections/MetersAccountsSection";
import { LeaseSection, type TenantLeaseValue } from "./tenant-sections/LeaseSection";
import { FacilityNoteSection } from "./tenant-sections/FacilityNoteSection";

const emptyMeters = (): MeterRow[] => [
  { utility: "water", meterNo: "", customerNo: "" }, { utility: "elec", meterNo: "", customerNo: "" }, { utility: "gas", meterNo: "", customerNo: "" },
];
const emptyAccounts = (): AccountRow[] => [
  { kind: "water", bank: "", account: "", holder: "", cmsMemberNo: "" }, { kind: "elec", bank: "", account: "", holder: "", cmsMemberNo: "" },
  { kind: "gas", bank: "", account: "", holder: "", cmsMemberNo: "" }, { kind: "mgmt", bank: "", account: "", holder: "", cmsMemberNo: "" },
];

export function TenantForm({ buildingId, unitId }: { buildingId: number; unitId?: number }) {
  const router = useRouter();
  const save = useSaveUnitOccupancy();
  const { data: roomTypes } = useRoomTypes();
  const { data: full } = useUnitOccupancyFull(unitId ?? 0);

  const [basic, setBasic] = useState<TenantBasicValue>({ unitNo: "", tenantName: "", payerName: "", phone: "", phone2: "", phone3: "", vehicleNo: "", residentCount: null, roomTypeId: null, moveInDate: "" });
  const [billing, setBilling] = useState<TenantBillingValue>({ isVacant: false, moveOutDate: "", billingEnabled: true, printEnabled: true, lateFeeEnabled: true, singlePrint: false, billingNote: "", settlementNote: "" });
  const [meters, setMeters] = useState<MeterRow[]>(emptyMeters());
  const [accounts, setAccounts] = useState<AccountRow[]>(emptyAccounts());
  const [lease, setLease] = useState<TenantLeaseValue>({ deposit: null, monthlyRent: null, startDate: "", endDate: "", expiryAlert: false });
  const [facilityNote, setFacilityNote] = useState("");
  const [occupancyId, setOccupancyId] = useState<number | null>(null);
  const [dirty, setDirty] = useState(false);
  const [errors, setErrors] = useState<Partial<Record<keyof TenantBasicValue, string>>>({});

  const roomTypeOptions = useMemo(
    () => [{ value: "", label: "선택" }, ...(roomTypes ?? []).map((r) => ({ value: String(r.id), label: r.label }))],
    [roomTypes],
  );

  useEffect(() => {
    if (!full) return;
    const o = full.occupancy;
    setBasic({
      unitNo: full.unit.unitNo, tenantName: o?.tenantName ?? "", payerName: o?.payerName ?? "", phone: o?.phone ?? "",
      phone2: o?.phone2 ?? "", phone3: o?.phone3 ?? "", vehicleNo: o?.vehicleNo ?? "", residentCount: o?.residentCount ?? null,
      roomTypeId: o?.roomTypeId ?? null, moveInDate: o?.moveInDate ?? "",
    });
    setBilling({
      isVacant: o?.status === "vacant", moveOutDate: o?.moveOutDate ?? "",
      billingEnabled: o?.billingEnabled ?? true, printEnabled: o?.printEnabled ?? true,
      lateFeeEnabled: o?.lateFeeEnabled ?? true, singlePrint: o?.singlePrint ?? false,
      billingNote: o?.billingNote ?? "", settlementNote: o?.settlementNote ?? "",
    });
    setFacilityNote(o?.facilityNote ?? "");
    setOccupancyId(o?.id ?? null);
    setMeters(emptyMeters().map((m) => { const f = full.meters.find((x) => x.utility === m.utility); return f ? { utility: m.utility, meterNo: f.meterNo ?? "", customerNo: f.customerNo ?? "" } : m; }));
    setAccounts(emptyAccounts().map((a) => { const f = full.accounts.find((x) => x.kind === a.kind); return f ? { kind: a.kind, bank: f.bank ?? "", account: f.account ?? "", holder: f.holder ?? "", cmsMemberNo: f.cmsMemberNo ?? "" } : a; }));
    setLease(full.lease ? { deposit: full.lease.deposit, monthlyRent: full.lease.monthlyRent, startDate: full.lease.startDate ?? "", endDate: full.lease.endDate ?? "", expiryAlert: full.lease.expiryAlert } : { deposit: null, monthlyRent: null, startDate: "", endDate: "", expiryAlert: false });
    setDirty(false);
  }, [full]);

  const mark = <T,>(setter: (v: T) => void) => (v: T) => { setter(v); setDirty(true); };
  const patchBasic = (p: Partial<TenantBasicValue>) => { setBasic((b) => ({ ...b, ...p })); setDirty(true); };
  const patchBilling = (p: Partial<TenantBillingValue>) => { setBilling((b) => ({ ...b, ...p })); setDirty(true); };

  const onSave = async () => {
    const parsed = occupancyBasicSchema.safeParse({
      unitNo: basic.unitNo, tenantName: basic.tenantName, payerName: basic.payerName, phone: basic.phone,
      phone2: basic.phone2, phone3: basic.phone3, vehicleNo: basic.vehicleNo, residentCount: basic.residentCount,
      roomTypeId: basic.roomTypeId ?? 0, moveInDate: basic.moveInDate,
    });
    if (!parsed.success) {
      const fe: Partial<Record<keyof TenantBasicValue, string>> = {};
      for (const issue of parsed.error.issues) { const k = issue.path[0] as keyof TenantBasicValue; if (!fe[k]) fe[k] = issue.message; }
      setErrors(fe); toast.error("필수 항목을 확인해 주세요."); return;
    }
    setErrors({});
    const payload: SaveUnitOccupancyPayload = {
      buildingId, unitId: unitId ?? null, occupancyId,
      unit: { unitNo: basic.unitNo },
      occupancy: {
        tenantName: basic.tenantName, payerName: basic.payerName, phone: basic.phone, phone2: basic.phone2, phone3: basic.phone3,
        vehicleNo: basic.vehicleNo, residentCount: basic.residentCount, roomTypeId: basic.roomTypeId, moveInDate: basic.moveInDate,
        moveOutDate: billing.moveOutDate || null, status: billing.isVacant ? "vacant" : "occupied",
        billingEnabled: billing.billingEnabled, printEnabled: billing.printEnabled, lateFeeEnabled: billing.lateFeeEnabled,
        singlePrint: billing.singlePrint, billingNote: billing.billingNote, settlementNote: billing.settlementNote, facilityNote,
      },
      meters, accounts,
      lease: { deposit: lease.deposit, monthlyRent: lease.monthlyRent, startDate: lease.startDate || null, endDate: lease.endDate || null, expiryAlert: lease.expiryAlert },
    };
    try {
      const res = await save.mutateAsync(payload);
      toast.success("저장되었습니다."); setDirty(false);
      if (!unitId) router.replace(`/staff/buildings/${buildingId}/tenants/${res.unitId}`);
    } catch (e) {
      toast.error(e instanceof Error ? e.message : "저장에 실패했습니다.");
    }
  };

  const onLeave = () => {
    if (dirty && !window.confirm("저장하지 않고 나가시겠습니까?")) return;
    router.push(`/staff/buildings/${buildingId}/tenants`);
  };

  return (
    <div className="flex flex-col gap-4">
      <div className="flex items-center justify-between">
        <h2 className="text-lg font-extrabold text-t1">{unitId ? `${basic.unitNo} · ${basic.tenantName || "입주자"}` : "입주자 등록"}</h2>
        <div className="flex gap-2">
          <Button variant="ghost" onClick={onLeave}>목록으로</Button>
          <Button variant="primary" onClick={onSave} disabled={save.isPending}>저장</Button>
        </div>
      </div>
      <BasicInfoSection value={basic} onChange={patchBasic} roomTypeOptions={roomTypeOptions} errors={errors} />
      <BillingSettingsSection value={billing} onChange={patchBilling} />
      <MetersAccountsSection meters={meters} accounts={accounts} onMeters={mark(setMeters)} onAccounts={mark(setAccounts)} />
      <LeaseSection value={lease} onChange={mark((v: Partial<TenantLeaseValue>) => setLease((l) => ({ ...l, ...v })))} history={full?.history ?? []} />
      <FacilityNoteSection value={facilityNote} onChange={mark(setFacilityNote)} />
    </div>
  );
}

import { createClient } from "@/lib/supabase/server";
import { rowsToXlsxBuffer } from "@/lib/export/xlsx";

export async function GET(request: Request, ctx: { params: Promise<{ id: string }> }) {
  const { id } = await ctx.params;
  const buildingId = Number(id);
  const sb = await createClient();
  const { data, error } = await sb.rpc("list_tenants", { p_building_id: buildingId });
  if (error) return new Response(JSON.stringify({ error: error.message }), { status: 500 });

  const table = (data ?? []).map((r: any) => ({
    unitNo: r.unit_no,
    tenant: r.is_vacant ? "공실" : (r.tenant_name ?? ""),
    roomType: r.room_type_label ?? "",
    resident: r.resident_count ?? "",
    moveIn: r.move_in_date ?? "",
  }));
  const buffer = await rowsToXlsxBuffer(
    [
      { header: "호실명", key: "unitNo", width: 12 }, { header: "입주자명", key: "tenant", width: 16 },
      { header: "룸타입", key: "roomType", width: 12 }, { header: "인원", key: "resident", width: 8 },
      { header: "입주일", key: "moveIn", width: 14 },
    ],
    table,
  );
  const d = new Date();
  const ymd = `${d.getFullYear()}${String(d.getMonth() + 1).padStart(2, "0")}${String(d.getDate()).padStart(2, "0")}`;
  return new Response(new Uint8Array(buffer), {
    status: 200,
    headers: {
      "Content-Type": "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
      "Content-Disposition": `attachment; filename="tenant-cards_${ymd}.xlsx"; filename*=UTF-8''${encodeURIComponent(`입주자카드_${ymd}.xlsx`)}`,
    },
  });
}
