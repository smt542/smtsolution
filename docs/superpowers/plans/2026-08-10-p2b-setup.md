> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `/staff/setup` 5탭(고지서·계좌·이사정산·시스템·장기수선비)을 순수 RLS CRUD로 구현하고, 민감 config 테이블에 첫 module-permission RLS를 부착한다.

**Architecture:** 단일 client 페이지 + 탭 상태. 신규 DB(`0003`)는 `setup_config`(단일행 upsert) + `settlement_check_items` + 마스터 5종 + `long_term_repair_history`. 데이터는 하우스 컨벤션의 reactquery 도메인(apis/keys/queries/mutations/index+apis.test), label 마스터는 공용 팩토리로 보일러플레이트 축소. 협력업체(vendors)는 기존 컴포넌트를 탭④로 편입하고 독립 라우트는 리다이렉트.

**Tech Stack:** Next.js 16(App Router) · Supabase(PostgreSQL·Auth, anon+RLS) · React Query v5 · zod · Tailwind v4 · Vitest 4 + Testing Library.

```sql
do $$ begin
  perform 1 from pg_policies where tablename='setup_config' and policyname='sc_update'
    and qual like '%current_user_has_module%';
  if not found then raise exception 'sc_update must gate by current_user_has_module'; end if;
end $$;

do $$ begin
  perform 1 from pg_policies where tablename='room_types' and policyname='rt_all'
    and coalesce(qual,'') not like '%current_user_has_module%';
  if not found then raise exception 'rt_all must NOT gate by module'; end if;
```

```ts
// hooks/reactquery/_shared/labelMaster.ts
import { createClient } from "@/lib/supabase/client";

export interface LabelMasterRow { id: number; orgId: number; label: string; sortOrder: number; }
export interface LabelMasterInput { orgId: number; label: string; sortOrder?: number; }

const toRow = (r: any): LabelMasterRow => ({ id: r.id, orgId: r.org_id, label: r.label, sortOrder: r.sort_order });

export function makeLabelMasterApis(table: string) {
  return {
    async list(): Promise<LabelMasterRow[]> {
      const { data, error } = await createClient().from(table).select("*").is("deleted_at", null).order("sort_order");
      if (error) throw new Error(error.message);
      return (data ?? []).map(toRow);
    },
    async create(input: LabelMasterInput): Promise<LabelMasterRow> {
      const row: Record<string, unknown> = { org_id: input.orgId, label: input.label };
      if (input.sortOrder !== undefined) row.sort_order = input.sortOrder;
      const { data, error } = await createClient().from(table).insert(row).select().single();
      if (error) throw new Error(error.message);
      return toRow(data);
    },
    async rename(id: number, label: string): Promise<LabelMasterRow> {
      const { data, error } = await createClient().from(table).update({ label }).eq("id", id).select().single();
      if (error) throw new Error(error.message);
      return toRow(data);
    },
    async softDelete(id: number): Promise<void> {
      const { error } = await createClient().from(table).update({ deleted_at: new Date().toISOString() }).eq("id", id);
      if (error) throw new Error(error.message);
    },
  };
}
```

```ts
export interface SetupConfig {
  orgId: number;
  customerNotice: string | null;
  unitLabelDefault: "code" | "name";
  lateFeeMerge: boolean;
  printUtilityCutoff: boolean;
  printWaterUnitPrice: boolean;
  printBillingPeriod: boolean;
  printDelta: boolean;
  showAsOnMain: boolean;
  cutoffMonths: number | null;
  mgmtBank1: string | null; mgmtAcct1: string | null; mgmtHolder1: string | null; mgmtPrint1: boolean;
  mgmtBank2: string | null; mgmtAcct2: string | null; mgmtHolder2: string | null; mgmtPrint2: boolean;
  settleBank: string | null; settleAcct: string | null; settleHolder: string | null;
  waterUnitReflect: boolean;
  electricUnitReflect: boolean;
  vacancyPaymentMode: "manual" | "auto";
}
export type SetupConfigPatch = Partial<Omit<SetupConfig, "orgId">>;

const rowToConfig = (r: any): SetupConfig => ({
  orgId: r.org_id, customerNotice: r.customer_notice, unitLabelDefault: r.unit_label_default,
  lateFeeMerge: r.late_fee_merge, printUtilityCutoff: r.print_utility_cutoff,
  printWaterUnitPrice: r.print_water_unit_price, printBillingPeriod: r.print_billing_period,
  printDelta: r.print_delta, showAsOnMain: r.show_as_on_main, cutoffMonths: r.cutoff_months,
  mgmtBank1: r.mgmt_bank1, mgmtAcct1: r.mgmt_acct1, mgmtHolder1: r.mgmt_holder1, mgmtPrint1: r.mgmt_print1,
  mgmtBank2: r.mgmt_bank2, mgmtAcct2: r.mgmt_acct2, mgmtHolder2: r.mgmt_holder2, mgmtPrint2: r.mgmt_print2,
  settleBank: r.settle_bank, settleAcct: r.settle_acct, settleHolder: r.settle_holder,
  waterUnitReflect: r.water_unit_reflect, electricUnitReflect: r.electric_unit_reflect,
  vacancyPaymentMode: r.vacancy_payment_mode,
});

const CAMEL_TO_SNAKE: Record<keyof SetupConfigPatch, string> = {
  customerNotice:"customer_notice", unitLabelDefault:"unit_label_default", lateFeeMerge:"late_fee_merge",
  printUtilityCutoff:"print_utility_cutoff", printWaterUnitPrice:"print_water_unit_price",
  printBillingPeriod:"print_billing_period", printDelta:"print_delta", showAsOnMain:"show_as_on_main",
  cutoffMonths:"cutoff_months", mgmtBank1:"mgmt_bank1", mgmtAcct1:"mgmt_acct1", mgmtHolder1:"mgmt_holder1",
  mgmtPrint1:"mgmt_print1", mgmtBank2:"mgmt_bank2", mgmtAcct2:"mgmt_acct2", mgmtHolder2:"mgmt_holder2",
  mgmtPrint2:"mgmt_print2", settleBank:"settle_bank", settleAcct:"settle_acct", settleHolder:"settle_holder",
  waterUnitReflect:"water_unit_reflect", electricUnitReflect:"electric_unit_reflect",
  vacancyPaymentMode:"vacancy_payment_mode",
};

function patchToRow(patch: SetupConfigPatch): Record<string, unknown> {
  const r: Record<string, unknown> = {};
  for (const k of Object.keys(patch) as (keyof SetupConfigPatch)[]) {
    if (patch[k] !== undefined) r[CAMEL_TO_SNAKE[k]] = patch[k];
  }
  return r;
}

export async function getSetupConfig(): Promise<SetupConfig | null> {
  const { data, error } = await createClient().from("setup_config").select("*").maybeSingle();
  if (error) throw new Error(error.message);
  return data ? rowToConfig(data) : null;
}

export async function upsertSetupConfig(orgId: number, patch: SetupConfigPatch): Promise<SetupConfig> {
  const { data, error } = await createClient()
    .from("setup_config")
    .upsert({ org_id: orgId, ...patchToRow(patch) }, { onConflict: "org_id" })
    .select().single();
  if (error) throw new Error(error.message);
  return rowToConfig(data);
}
```

```tsx
// app/(staff)/staff/setup/components/MasterListEditor.tsx
"use client";
import { useState } from "react";
import { Button, Input } from "@/components/common";

export interface MasterItem { id: number; label: string; }

export function MasterListEditor({
  title, items, onAdd, onRename, onDelete,
}: {
  title: string;
  items: MasterItem[];
  onAdd: (label: string) => void;
  onRename: (id: number, label: string) => void;
  onDelete: (id: number) => void;
}) {
  const [newLabel, setNewLabel] = useState("");
  const [editingId, setEditingId] = useState<number | null>(null);
  const [editLabel, setEditLabel] = useState("");

  const add = () => {
    const v = newLabel.trim();
    if (!v) return;
    onAdd(v);
    setNewLabel("");
  };
  const saveEdit = () => {
    const v = editLabel.trim();
    if (editingId != null && v) onRename(editingId, v);
    setEditingId(null);
    setEditLabel("");
  };

  return (
    <div className="rounded-2xl border border-line bg-surface p-4">
      <h4 className="mb-3 text-sm font-extrabold text-t1">{title}</h4>
      <ul className="flex flex-col gap-1">
        {items.length === 0 && <li className="py-2 text-sm text-t4">등록된 항목이 없습니다.</li>}
        {items.map((it) => (
          <li key={it.id} className="flex items-center gap-2">
            {editingId === it.id ? (
              <>
                <Input
                  aria-label={`${title} 이름 변경`}
                  value={editLabel}
                  onChange={(e) => setEditLabel(e.target.value)}
                  onKeyDown={(e) => { if (e.key === "Enter") saveEdit(); if (e.key === "Escape") setEditingId(null); }}
                />
                <Button variant="primary" onClick={saveEdit}>저장</Button>
                <button onClick={() => setEditingId(null)} className="text-xs text-t3">취소</button>
              </>
            ) : (
              <>
                <span className="flex-1 text-sm text-t2">{it.label}</span>
                <button aria-label={`${it.label} 수정`} onClick={() => { setEditingId(it.id); setEditLabel(it.label); }} className="text-xs text-t3 hover:text-brand">수정</button>
                <button aria-label={`${it.label} 삭제`} onClick={() => onDelete(it.id)} className="text-xs text-t3 hover:text-danger">삭제</button>
              </>
            )}
          </li>
        ))}
      </ul>
      <div className="mt-3 flex items-center gap-2">
        <Input
          aria-label={`${title} 추가`}
          placeholder="새 항목"
          value={newLabel}
          onChange={(e) => setNewLabel(e.target.value)}
          onKeyDown={(e) => { if (e.key === "Enter") add(); }}
        />
        <Button variant="primary" onClick={add}>＋ 추가</Button>
      </div>
    </div>
  );
}
```

```tsx
// app/(staff)/staff/setup/components/MovingSettlementSettingsTab.tsx
"use client";
import { useEffect, useState } from "react";
import { toast } from "sonner";
import { Button, Select, Checkbox, Panel } from "@/components/common";
import { useMe } from "@/hooks/reactquery/auth";
import { useSetupConfig, useUpsertSetupConfig } from "@/hooks/reactquery/setupConfig";
import {
  useSettlementCheckItems, useCreateSettlementCheckItem,
  useRenameSettlementCheckItem, useDeleteSettlementCheckItem,
} from "@/hooks/reactquery/settlementCheckItems";
import { MasterListEditor } from "./MasterListEditor";

export function MovingSettlementSettingsTab() {
  const { data: me } = useMe();
  const orgId = me?.orgId;
  const { data: config } = useSetupConfig();
  const upsert = useUpsertSetupConfig();

  const [waterUnitReflect, setWater] = useState(false);
  const [electricUnitReflect, setElec] = useState(false);
  const [vacancyPaymentMode, setMode] = useState<"manual" | "auto">("manual");
  useEffect(() => {
    if (!config) return;
    setWater(config.waterUnitReflect);
    setElec(config.electricUnitReflect);
    setMode(config.vacancyPaymentMode);
  }, [config]);

  const save = async () => {
    if (!orgId) { toast.error("인증 정보가 없습니다"); return; }
    try {
      await upsert.mutateAsync({ orgId, patch: { waterUnitReflect, electricUnitReflect, vacancyPaymentMode } });
      toast.success("저장되었습니다");
    } catch (e: any) { toast.error(e.message ?? "저장 실패"); }
  };

  const { data: items = [] } = useSettlementCheckItems();
  const createItem = useCreateSettlementCheckItem();
  const renameItem = useRenameSettlementCheckItem();
  const deleteItem = useDeleteSettlementCheckItem();
  const onAdd = async (label: string) => {
    if (!orgId) return;
    try { await createItem.mutateAsync({ orgId, label }); } catch (e: any) { toast.error(e.message ?? "추가 실패"); }
  };
  const onRename = async (id: number, label: string) => {
    try { await renameItem.mutateAsync({ id, label }); } catch (e: any) { toast.error(e.message ?? "수정 실패"); }
  };
  const onDelete = async (id: number) => {
    try { await deleteItem.mutateAsync(id); } catch (e: any) { toast.error(e.message ?? "삭제 실패"); }
  };

  return (
    <div className="flex flex-col gap-6">
      <Panel title="이사정산 설정" action={<Button variant="primary" onClick={save} disabled={upsert.isPending}>저장</Button>}>
        <div className="flex flex-col gap-4">
          <Checkbox label="이사정산 시 수도 단가 반영" checked={waterUnitReflect} onChange={(e) => setWater(e.target.checked)} />
          <Checkbox label="이사정산 시 전기 단가 반영" checked={electricUnitReflect} onChange={(e) => setElec(e.target.checked)} />
          <label className="flex flex-col gap-1">
            <span className="text-sm text-t2">공실 처리 시 수납 방식</span>
            <Select aria-label="공실 처리 시 수납 방식" value={vacancyPaymentMode}
              onChange={(e) => setMode(e.target.value as "manual" | "auto")}
              options={[{ value: "manual", label: "수동" }, { value: "auto", label: "자동" }]} />
          </label>
        </div>
      </Panel>
      <MasterListEditor
        title="이사정산 체크항목"
        items={items.map((i) => ({ id: i.id, label: i.label }))}
        onAdd={onAdd} onRename={onRename} onDelete={onDelete}
      />
    </div>
  );
}
```

```ts
// hooks/reactquery/_shared/labelMasterHooks.ts
import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";
import type { LabelMasterInput, LabelMasterRow } from "./labelMaster";

export interface LabelMasterApi {
  list: () => Promise<LabelMasterRow[]>;
  create: (i: LabelMasterInput) => Promise<LabelMasterRow>;
  rename: (id: number, label: string) => Promise<LabelMasterRow>;
  softDelete: (id: number) => Promise<void>;
}

export function makeLabelMasterHooks(base: string, api: LabelMasterApi) {
  const keys = { all: [base] as const, list: () => [base, "list"] as const };
  return {
    keys,
    useList: () => useQuery({ queryKey: keys.list(), queryFn: api.list }),
    useCreate: () => {
      const qc = useQueryClient();
      return useMutation({ mutationFn: (i: LabelMasterInput) => api.create(i), onSuccess: () => qc.invalidateQueries({ queryKey: keys.all }) });
    },
    useRename: () => {
      const qc = useQueryClient();
      return useMutation({ mutationFn: ({ id, label }: { id: number; label: string }) => api.rename(id, label), onSuccess: () => qc.invalidateQueries({ queryKey: keys.all }) });
    },
    useRemove: () => {
      const qc = useQueryClient();
      return useMutation({ mutationFn: (id: number) => api.softDelete(id), onSuccess: () => qc.invalidateQueries({ queryKey: keys.all }) });
    },
  };
}
```

```ts
// hooks/reactquery/quickMessages/apis.ts
import { createClient } from "@/lib/supabase/client";

export interface QuickMessage { id: number; orgId: number; title: string; body: string; sortOrder: number; }
export interface QuickMessageInput { orgId: number; title: string; body: string; sortOrder?: number; }

const toRow = (r: any): QuickMessage => ({ id: r.id, orgId: r.org_id, title: r.title, body: r.body, sortOrder: r.sort_order });

function inputToRow(i: Partial<QuickMessageInput>): Record<string, unknown> {
  const r: Record<string, unknown> = {};
  const set = (k: string, v: unknown) => { if (v !== undefined) r[k] = v; };
  set("org_id", i.orgId); set("title", i.title); set("body", i.body); set("sort_order", i.sortOrder);
  return r;
}

export async function listQuickMessages(): Promise<QuickMessage[]> {
  const { data, error } = await createClient().from("quick_messages").select("*").is("deleted_at", null).order("sort_order");
  if (error) throw new Error(error.message);
  return (data ?? []).map(toRow);
}
export async function createQuickMessage(input: QuickMessageInput): Promise<QuickMessage> {
  const { data, error } = await createClient().from("quick_messages").insert(inputToRow(input)).select().single();
  if (error) throw new Error(error.message);
  return toRow(data);
}
export async function updateQuickMessage(id: number, input: Partial<QuickMessageInput>): Promise<QuickMessage> {
  const { data, error } = await createClient().from("quick_messages").update(inputToRow(input)).eq("id", id).select().single();
  if (error) throw new Error(error.message);
  return toRow(data);
}
export async function deleteQuickMessage(id: number): Promise<void> {
  const { error } = await createClient().from("quick_messages").update({ deleted_at: new Date().toISOString() }).eq("id", id);
  if (error) throw new Error(error.message);
}
```

```tsx
// app/(staff)/staff/setup/components/QuickMessageEditor.tsx
"use client";
import { useState } from "react";
import { toast } from "sonner";
import { Button, Input, Modal } from "@/components/common";
import { useMe } from "@/hooks/reactquery/auth";
import {
  useQuickMessages, useCreateQuickMessage, useUpdateQuickMessage, useDeleteQuickMessage,
  type QuickMessage,
} from "@/hooks/reactquery/quickMessages";

export function QuickMessageEditor() {
  const { data: me } = useMe();
  const orgId = me?.orgId;
  const { data: messages = [] } = useQuickMessages();
  const create = useCreateQuickMessage();
  const update = useUpdateQuickMessage();
  const del = useDeleteQuickMessage();

  const [open, setOpen] = useState(false);
  const [editing, setEditing] = useState<QuickMessage | null>(null);
  const [title, setTitle] = useState("");
  const [body, setBody] = useState("");

  const openCreate = () => { setEditing(null); setTitle(""); setBody(""); setOpen(true); };
  const openEdit = (m: QuickMessage) => { setEditing(m); setTitle(m.title); setBody(m.body); setOpen(true); };

  const submit = async () => {
    if (!title.trim() || !body.trim()) { toast.error("제목과 내용을 입력해 주세요"); return; }
    try {
      if (editing) await update.mutateAsync({ id: editing.id, input: { title, body } });
      else {
        if (!orgId) { toast.error("인증 정보가 없습니다"); return; }
        await create.mutateAsync({ orgId, title, body });
      }
      setOpen(false);
    } catch (e: any) { toast.error(e.message ?? "저장 실패"); }
  };
  const remove = async (id: number) => {
    try { await del.mutateAsync(id); } catch (e: any) { toast.error(e.message ?? "삭제 실패"); }
  };

  return (
    <div className="rounded-2xl border border-line bg-surface p-4">
      <div className="mb-3 flex items-center justify-between">
        <h4 className="text-sm font-extrabold text-t1">자주 보내는 문자</h4>
        <Button variant="primary" onClick={openCreate}>＋ 추가</Button>
      </div>
      <ul className="flex flex-col gap-1">
        {messages.length === 0 && <li className="py-2 text-sm text-t4">등록된 문자가 없습니다.</li>}
        {messages.map((m) => (
          <li key={m.id} className="flex items-center gap-2">
            <span className="flex-1 truncate text-sm text-t2"><b>{m.title}</b> — {m.body}</span>
            <button aria-label={`${m.title} 수정`} onClick={() => openEdit(m)} className="text-xs text-t3 hover:text-brand">수정</button>
            <button aria-label={`${m.title} 삭제`} onClick={() => remove(m.id)} className="text-xs text-t3 hover:text-danger">삭제</button>
          </li>
        ))}
      </ul>
      <Modal open={open} onClose={() => setOpen(false)} title={editing ? "문자 수정" : "문자 추가"}>
```

```ts
// hooks/reactquery/longTermRepair/apis.ts
import { createClient } from "@/lib/supabase/client";

export interface LongTermRepair { id: number; orgId: number; buildingId: number; workName: string; workDate: string | null; amount: number | null; memo: string | null; }
export interface LongTermRepairInput { orgId: number; buildingId: number; workName: string; workDate?: string | null; amount?: number | null; memo?: string | null; }
export interface RepairAttachment { id: number; orgId: number; entityType: string; entityId: number; kind: string; fileUrl: string; fileName: string; createdAt: string; }

const toRepair = (r: any): LongTermRepair => ({ id: r.id, orgId: r.org_id, buildingId: r.building_id, workName: r.work_name, workDate: r.work_date, amount: r.amount, memo: r.memo });
const toAtt = (r: any): RepairAttachment => ({ id: r.id, orgId: r.org_id, entityType: r.entity_type, entityId: r.entity_id, kind: r.kind, fileUrl: r.file_url, fileName: r.file_name, createdAt: r.created_at });

function inputToRow(i: Partial<LongTermRepairInput>): Record<string, unknown> {
  const r: Record<string, unknown> = {};
  const set = (k: string, v: unknown) => { if (v !== undefined) r[k] = v; };
  set("org_id", i.orgId); set("building_id", i.buildingId); set("work_name", i.workName);
  set("work_date", i.workDate); set("amount", i.amount); set("memo", i.memo);
  return r;
}

export async function listRepairsByBuilding(buildingId: number): Promise<LongTermRepair[]> {
  const { data, error } = await createClient().from("long_term_repair_history").select("*")
    .eq("building_id", buildingId).is("deleted_at", null).order("work_date", { ascending: false });
  if (error) throw new Error(error.message);
  return (data ?? []).map(toRepair);
}
export async function createRepair(input: LongTermRepairInput): Promise<LongTermRepair> {
  const { data, error } = await createClient().from("long_term_repair_history").insert(inputToRow(input)).select().single();
  if (error) throw new Error(error.message);
  return toRepair(data);
}
export async function updateRepair(id: number, input: Partial<LongTermRepairInput>): Promise<LongTermRepair> {
  const { data, error } = await createClient().from("long_term_repair_history").update(inputToRow(input)).eq("id", id).select().single();
  if (error) throw new Error(error.message);
  return toRepair(data);
}
export async function deleteRepair(id: number): Promise<void> {
  const { error } = await createClient().from("long_term_repair_history").update({ deleted_at: new Date().toISOString() }).eq("id", id);
  if (error) throw new Error(error.message);
}

export async function uploadRepairAttachment({ repairId, orgId, file }: { repairId: number; orgId: number; file: File }): Promise<RepairAttachment> {
  const sb = createClient();
  const path = `${orgId}/long_term_repair/${repairId}/${crypto.randomUUID()}-${file.name}`;
  const { error: upErr } = await sb.storage.from("attachments").upload(path, file);
  if (upErr) throw new Error(upErr.message);
  const { data, error } = await sb.from("attachments")
    .insert({ org_id: orgId, entity_type: "long_term_repair", entity_id: repairId, kind: "repair", file_url: path, file_name: file.name })
    .select().single();
  if (error) { await sb.storage.from("attachments").remove([path]); throw new Error(error.message); }
  return toAtt(data);
}
export async function listRepairAttachments(repairId: number): Promise<RepairAttachment[]> {
  const { data, error } = await createClient().from("attachments").select("*").eq("entity_type", "long_term_repair").eq("entity_id", repairId);
  if (error) throw new Error(error.message);
  return (data ?? []).map(toAtt);
}
export async function deleteRepairAttachment({ id, path }: { id: number; path: string }): Promise<void> {
  const sb = createClient();
  const { error: sErr } = await sb.storage.from("attachments").remove([path]);
  if (sErr) throw new Error(sErr.message);
  const { error } = await sb.from("attachments").delete().eq("id", id);
  if (error) throw new Error(error.message);
}
export async function getRepairAttachmentSignedUrl(path: string, expiresInSec = 3600): Promise<string> {
  const { data, error } = await createClient().storage.from("attachments").createSignedUrl(path, expiresInSec);
  if (error) throw new Error(error.message);
  return data.signedUrl;
}
```

```tsx
"use client";
import React, { useState } from "react";
import { toast } from "sonner";
import { Button, Select, Input, Modal } from "@/components/common";
import { useMe } from "@/hooks/reactquery/auth";
import { useBuildings } from "@/hooks/reactquery/buildings";

  useRepairsByBuilding, useCreateRepair, useUpdateRepair, useDeleteRepair,
  useRepairAttachments, useUploadRepairAttachment, useDeleteRepairAttachment,
  type LongTermRepair, type RepairAttachment,
} from "@/hooks/reactquery/longTermRepair";

function AttachmentLoader({ repairId, children }: {
  repairId: number;
  children: (d: { attachments: RepairAttachment[]; onUpload: (f: File) => Promise<void>; onDelete: (a: RepairAttachment) => Promise<void> }) => React.ReactNode;
}) {
  const { data: me } = useMe();
  const orgId = me?.orgId;
  const { data: attachments = [] } = useRepairAttachments(repairId);
  const upload = useUploadRepairAttachment();
  const del = useDeleteRepairAttachment();
  const onUpload = async (file: File) => {
    if (!orgId) { toast.error("인증 정보가 없습니다"); return; }
    try { await upload.mutateAsync({ repairId, orgId, file }); } catch (e: any) { toast.error(e.message ?? "업로드 실패"); }
  };
  const onDelete = async (a: RepairAttachment) => {
    try { await del.mutateAsync({ id: a.id, path: a.fileUrl, repairId }); } catch (e: any) { toast.error(e.message ?? "삭제 실패"); }
  };
  return <>{children({ attachments, onUpload, onDelete })}</>;
}

export function LongTermRepairTab() {
  const { data: me } = useMe();
  const orgId = me?.orgId;
  const { data: buildings = [] } = useBuildings();
  const [buildingId, setBuildingId] = useState<number | null>(null);
  const { data: repairs = [] } = useRepairsByBuilding(buildingId ?? undefined);

  const create = useCreateRepair();
  const update = useUpdateRepair();
  const del = useDeleteRepair();

  const [open, setOpen] = useState(false);
  const [editing, setEditing] = useState<LongTermRepair | null>(null);
  const [workName, setWorkName] = useState("");
  const [workDate, setWorkDate] = useState("");
  const [amount, setAmount] = useState("");
  const [memo, setMemo] = useState("");

  const openCreate = () => { setEditing(null); setWorkName(""); setWorkDate(""); setAmount(""); setMemo(""); setOpen(true); };
  const openEdit = (r: LongTermRepair) => {
    setEditing(r); setWorkName(r.workName); setWorkDate(r.workDate ?? "");
    setAmount(r.amount != null ? String(r.amount) : ""); setMemo(r.memo ?? ""); setOpen(true);
  };

  const submit = async () => {
    if (!workName.trim()) { toast.error("공사명을 입력해 주세요"); return; }
    const payload = { workName, workDate: workDate || null, amount: amount === "" ? null : Number(amount), memo: memo || null };
    try {
      if (editing) await update.mutateAsync({ id: editing.id, input: payload });
      else {
        if (!orgId || !buildingId) { toast.error("건물을 선택해 주세요"); return; }
        await create.mutateAsync({ orgId, buildingId, ...payload });
      }
      setOpen(false);
    } catch (e: any) { toast.error(e.message ?? "저장 실패"); }
  };
  const remove = async (id: number) => { try { await del.mutateAsync(id); } catch (e: any) { toast.error(e.message ?? "삭제 실패"); } };

  return (
    <div className="flex flex-col gap-4">
      <div className="flex items-center justify-between">
        <label className="flex items-center gap-2">
          <span className="text-sm text-t2">건물</span>
          <Select aria-label="건물 선택" value={buildingId != null ? String(buildingId) : ""}
            onChange={(e) => setBuildingId(e.target.value ? Number(e.target.value) : null)}
            options={[{ value: "", label: "건물을 선택하세요" }, ...buildings.map((b) => ({ value: String(b.id), label: b.name }))]} />
        </label>
        <Button variant="primary" onClick={openCreate} disabled={!buildingId}>＋ 이력 추가</Button>
      </div>

      {!buildingId ? (
        <p className="text-sm text-t4">건물을 선택하면 장기수선비 이력이 표시됩니다.</p>
      ) : repairs.length === 0 ? (
        <p className="text-sm text-t4">등록된 이력이 없습니다.</p>
      ) : (
        <ul className="flex flex-col gap-1">
          {repairs.map((r) => (
            <li key={r.id} className="flex items-center gap-3 rounded-xl border border-line p-3">
              <span className="flex-1 text-sm text-t1">{r.workName}</span>
              <span className="text-sm text-t3">{r.workDate ?? "-"}</span>
              <span className="text-sm text-t2">{r.amount != null ? r.amount.toLocaleString() + "원" : "-"}</span>
              <button aria-label={`${r.workName} 수정`} onClick={() => openEdit(r)} className="text-xs text-t3 hover:text-brand">수정</button>
              <button aria-label={`${r.workName} 삭제`} onClick={() => remove(r.id)} className="text-xs text-t3 hover:text-danger">삭제</button>
            </li>
          ))}
        </ul>
      )}

      <Modal open={open} onClose={() => setOpen(false)} title={editing ? "장기수선비 수정" : "장기수선비 추가"}>
        <div className="flex flex-col gap-3">
          <Input aria-label="공사명" placeholder="공사명" value={workName} onChange={(e) => setWorkName(e.target.value)} />
          <Input aria-label="공사일" type="date" value={workDate} onChange={(e) => setWorkDate(e.target.value)} />
          <Input aria-label="금액" type="number" placeholder="금액(원)" value={amount} onChange={(e) => setAmount(e.target.value)} />
          <textarea aria-label="비고" placeholder="비고" value={memo} onChange={(e) => setMemo(e.target.value)}
            className="min-h-20 w-full rounded-xl border border-line bg-surface p-3 text-sm text-t1" />
          {editing && (
            <AttachmentLoader repairId={editing.id}>
              {({ attachments, onUpload, onDelete }) => (
                <div className="flex flex-col gap-2">
                  <span className="text-sm font-bold text-t2">영수증·사진</span>
                  <input type="file" accept="image/*,application/pdf" aria-label="파일 업로드"
                    onChange={(e) => { const f = e.target.files?.[0]; if (f) onUpload(f); }} />
                  <ul className="flex flex-col gap-1">
                    {attachments.map((a) => (
                      <li key={a.id} className="flex items-center gap-2 text-sm">
                        <span className="flex-1 truncate">{a.fileName}</span>
                        <button aria-label={`${a.fileName} 삭제`} onClick={() => onDelete(a)} className="text-xs text-t3 hover:text-danger">삭제</button>
                      </li>
                    ))}
                  </ul>
                </div>
              )}
            </AttachmentLoader>
          )}
          <div className="flex justify-end gap-2">
            <button onClick={() => setOpen(false)} className="text-sm text-t3">취소</button>
            <Button variant="primary" onClick={submit} disabled={create.isPending || update.isPending}>{editing ? "저장" : "추가"}</Button>
          </div>
        </div>
      </Modal>
    </div>
  );
}
```
